"use client";

import Link from "next/link";
import { useEffect, useRef, useState } from "react";
import "./allen-mega-navigation.css";
import "./aa-dynamic-overflow.css";

import { navigationGroups, type SessionNavigation } from "../lib/navigation";

export default function MainNavigation({ path, session }: { path: string; session: SessionNavigation }) {
  const groups = navigationGroups(session);
  const [mobileOpen, setMobileOpen] = useState(false);
  const [menuTop, setMenuTop] = useState(80);
  const [open, setOpen] = useState<string | null>(null);
  const timer = useRef<ReturnType<typeof setTimeout> | null>(null);
  const root = useRef<HTMLDivElement>(null);
  const probe = useRef<HTMLDivElement>(null);
  const [visibleGroups, setVisibleGroups] = useState(groups.length);
  const [moreOpen, setMoreOpen] = useState(false);

  useEffect(() => {
    const node = root.current;
    const measurement = probe.current;
    if (!node || !measurement) return;
    const calculate = () => {
      const widths = Array.from(measurement.children).map(el =>
        el.getBoundingClientRect().width
      );
      const gap = 8;
      const available = node.clientWidth;
      const full = widths.reduce((a,b) => a + b, 0) + gap * (widths.length - 1);
      if (full <= available) {
        setVisibleGroups(groups.length);
        return;
      }
      const budget = Math.max(0, available - 62);
      let used = widths[0] || 0;
      let count = 0;
      for (let i = 0; i < groups.length; i++) {
        const next = used + gap + (widths[i + 1] || 0);
        if (next > budget) break;
        used = next;
        count++;
      }
      setVisibleGroups(count);
    };
    const observer = new ResizeObserver(calculate);
    observer.observe(node);
    window.addEventListener("resize", calculate);
    document.fonts?.ready.then(calculate).catch(() => {});
    calculate();
    return () => {
      observer.disconnect();
      window.removeEventListener("resize", calculate);
    };
  }, [groups.length]);

  const hiddenGroups = groups.slice(visibleGroups);

  function cancel() {
    if (timer.current !== null) {
      clearTimeout(timer.current);
      timer.current = null;
    }
  }

  function close() {
    cancel();
    setOpen(null);
    setMobileOpen(false);
    setMoreOpen(false);
  }

  function desktop() {
    return window.matchMedia(
      "(min-width: 1001px) and (hover: hover) and (pointer: fine)"
    ).matches;
  }

  useEffect(() => {
    function outside(e: PointerEvent) {
      if (e.target instanceof Node && !root.current?.contains(e.target)) {
        if (timer.current !== null) clearTimeout(timer.current);
        timer.current = null;
        setOpen(null);
        setMobileOpen(false);
        setMoreOpen(false);
      }
    }

    document.addEventListener("pointerdown", outside);
    return () => {
      document.removeEventListener("pointerdown", outside);
      if (timer.current !== null) clearTimeout(timer.current);
    };
  }, []);

  return (
    <div
      className="navigation-root aa-v9-root aa-dyn-root"
      ref={root}
      onKeyDown={(e) => {
        if (e.key === "Escape") {
          e.preventDefault();
          const target = root.current?.querySelector<HTMLButtonElement>(mobileOpen ? ".aa-v9-toggle" : '.aa-v9-trigger[aria-expanded="true"]');
          close();
          target?.focus();
        }
      }}
      onBlur={(e) => {
        if (!e.currentTarget.contains(e.relatedTarget)) close();
      }}
    >
      <button
        className="aa-v9-toggle"
        type="button"
        aria-expanded={mobileOpen}
        aria-controls="aa-v9-nav"
        onClick={() => {
          cancel();
          setOpen(null);
          setMenuTop(root.current?.closest("header")?.getBoundingClientRect().bottom ?? 80);
          setMobileOpen(!mobileOpen);
        }}
      >
        {mobileOpen ? "Close" : "Menu"}
      </button>

      <nav
        id="aa-v9-nav"
        style={mobileOpen ? { top: menuTop, maxHeight: `calc(100dvh - ${menuTop + 12}px)` } : undefined}
        className={`aa-v9-nav aa-dyn-nav${mobileOpen ? " is-open" : ""}`}
        aria-label="Main navigation"
      >
        <Link href="/" className="aa-v9-link" onClick={close}>
          Home
        </Link>

        {groups.slice(0, visibleGroups).map((group) => {
          const active = group.links.some(([href]) =>
            path === href || (href === "/doctor-dashboard" && path.startsWith("/doctor-dashboard/"))
          );
          const expanded = open === group.id;

          return (
            <div
              className="aa-v9-group"
              key={group.id}
              onPointerEnter={(e) => {
                if (e.pointerType !== "mouse" || !desktop()) return;
                cancel();
                timer.current = setTimeout(() => {
                  setOpen(group.id);
                  timer.current = null;
                }, 300);
              }}
              onPointerLeave={(e) => {
                if (e.pointerType !== "mouse" || !desktop()) return;
                cancel();
                timer.current = setTimeout(() => {
                  setOpen((current) => current === group.id ? null : current);
                  timer.current = null;
                }, 300);
              }}
            >
              <button
                type="button"
                className={`aa-v9-trigger${expanded || active ? " is-active" : ""}`}
                aria-expanded={expanded}
                aria-controls={`aa-v9-panel-${group.id}`}
                onKeyDown={(e) => {
                  if (e.key === "ArrowDown") {
                    e.preventDefault(); cancel(); setOpen(group.id);
                    requestAnimationFrame(() => document.getElementById(`aa-v9-panel-${group.id}`)?.querySelector("a")?.focus());
                  }
                }}
                onClick={() => {
                  cancel();
                  setOpen(expanded ? null : group.id);
                }}
              >
                {group.label}
                <svg aria-hidden="true" width="14" height="14" viewBox="0 0 24 24" fill="none" stroke="currentColor" strokeWidth="2"><path d="m6 9 6 6 6-6" /></svg>
              </button>

              <div
                id={`aa-v9-panel-${group.id}`}
                className="aa-v9-dropdown"
                hidden={!expanded}
              >
                <h2>{group.label}</h2>
                {group.links.map(([href, label]) => (
                  <Link href={href} key={href} onClick={close}>
                    {label}
                  </Link>
                ))}
                {group.id === "account" && <>
                  {session.doctor && <form action="/api/doctor-logout" method="post"><button className="aa-logout" type="submit">Doctor Logout</button></form>}
                  {session.clinic && <form action="/api/clinic/auth/logout" method="post"><button className="aa-logout" type="submit">Clinic Logout</button></form>}
                  {session.hr && <form action="/api/hr/logout" method="post"><button className="aa-logout" type="submit">HR Logout</button></form>}
                </>}
              </div>
            </div>
          );
        })}

        {hiddenGroups.length > 0 && (
          <div className="aa-dyn-more-wrap">
            <button
              type="button"
              className="aa-dyn-more-button aa-v9-trigger"
              aria-label="More navigation"
              aria-expanded={moreOpen}
              aria-controls="aa-dyn-overflow-panel"
              onClick={() => { cancel(); setOpen(null); setMoreOpen(v => !v); }}
            >
              â˜° <span className="aa-dyn-more-label">More</span>
            </button>
            {moreOpen && (
              <div className="aa-dyn-more-panel" id="aa-dyn-overflow-panel">
                {hiddenGroups.map(group => (
                  <details className="aa-dyn-more-group" key={group.id}>
                    <summary>{group.label} <span aria-hidden="true">â–¾</span></summary>
                    <div className="aa-dyn-more-links">
                      {group.links.map(([href, label]) => (
                        <Link key={href} href={href} onClick={close}
                          aria-current={path === href ? "page" : undefined}>
                          {label}
                        </Link>
                      ))}
                      {group.id === "account" && <>
                        {session.doctor && <form action="/api/doctor-logout" method="post"><button type="submit">Doctor Logout</button></form>}
                        {session.clinic && <form action="/api/clinic/auth/logout" method="post"><button type="submit">Clinic Logout</button></form>}
                        {session.hr && <form action="/api/hr/logout" method="post"><button type="submit">HR Logout</button></form>}
                      </>}
                    </div>
                  </details>
                ))}
              </div>
            )}
          </div>
        )}
      </nav>
      <div ref={probe} className="aa-dyn-measure" aria-hidden="true">
        <span className="aa-v9-link">Home</span>
        {groups.map(group => (
          <button type="button" tabIndex={-1} key={group.id} className="aa-v9-trigger">
            {group.label} <svg aria-hidden="true" width="14" height="14" viewBox="0 0 24 24" fill="none" stroke="currentColor" strokeWidth="2"><path d="m6 9 6 6 6-6" /></svg>
          </button>
        ))}
      </div>
    </div>
  );
}
